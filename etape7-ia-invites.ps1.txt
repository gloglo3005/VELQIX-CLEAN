<#
  VelQix - ETAPE 7 : assistant IA accessible aux invites (corrige le 401 sur /api/ai/chat).

  Cause : /api/ai/chat exigeait un token (authMiddleware) -> un visiteur non connecte recevait 401
  et l'app affichait un faux "Aucun bien trouve". La route utilise maintenant optionalAuth :
  utilisateur connecte = limite par compte ; invite = limite par IP (meme aiLimiter).

  Fichiers touches (et seulement ceux-la), dans le dossier backend :
    src\index.ts
    src\routes\ai.ts
    src\middlewares\rateLimit.ts   (commentaire uniquement)

  Lancer :
    powershell -ExecutionPolicy Bypass -File .\etape7-ia-invites.ps1 -ApiRoot "C:\chemin\backend"

  Annuler :
    powershell -ExecutionPolicy Bypass -File .\etape7-ia-invites.ps1 -ApiRoot "C:\chemin\backend" -Rollback
#>
param(
  [Parameter(Mandatory = $true)][string]$ApiRoot,
  [switch]$Rollback
)

$ErrorActionPreference = 'Stop'
$Tag = 'bak-step7'

function Fail($msg) { Write-Host "ERREUR : $msg" -ForegroundColor Red; exit 1 }
function LF([string]$s) { return ($s -replace "`r`n", "`n") }

if (-not (Test-Path (Join-Path $ApiRoot 'src\index.ts'))) { Fail "ApiRoot invalide (src\index.ts introuvable) : $ApiRoot" }

$idx = Join-Path $ApiRoot 'src\index.ts'
$ai  = Join-Path $ApiRoot 'src\routes\ai.ts'
$rl  = Join-Path $ApiRoot 'src\middlewares\rateLimit.ts'
$Files = @($idx, $ai, $rl)

if ($Rollback) {
  foreach ($f in $Files) {
    $bak = "$f.$Tag"
    if (Test-Path $bak) { Copy-Item $bak $f -Force; Remove-Item $bak -Force; Write-Host "Restaure : $f" -ForegroundColor Green }
    else { Write-Host "Pas de sauvegarde pour : $f (ignore)" -ForegroundColor Yellow }
  }
  Write-Host "Rollback etape 7 termine." -ForegroundColor Cyan
  exit 0
}

$Edits = @()
function AddEdit($file, $old, $new, $count = 1) {
  $script:Edits += [pscustomobject]@{ File = $file; Old = (LF $old); New = (LF $new); Count = $count }
}

$o = @'
import { authMiddleware } from '../middlewares/authMiddleware';
'@
$n = @'
import { optionalAuth } from '../middlewares/authMiddleware';
'@
AddEdit $ai $o $n

$o = @'
router.post('/chat', authMiddleware, async
'@
$n = @'
router.post('/chat', optionalAuth, async
'@
AddEdit $ai $o $n

$o = @'
import { authMiddleware } from "./middlewares/authMiddleware";
'@
$n = @'
import { optionalAuth } from "./middlewares/authMiddleware";
'@
AddEdit $idx $o $n

$o = @'
app.use("/api/ai/chat",       authMiddleware, aiLimiter);
'@
$n = @'
// L'assistant est ouvert aux invites : limite par compte si connecte, sinon par IP.
app.use("/api/ai/chat",       optionalAuth, aiLimiter);
'@
AddEdit $idx $o $n

$o = @'
// Assistant IA : chaque appel coûte de l'argent (Gemini).
// À monter APRÈS authMiddleware pour que req.userId soit connu.
'@
$n = @'
// Assistant IA : chaque appel coûte de l'argent (Groq).
// À monter APRÈS authMiddleware / optionalAuth pour que req.userId soit connu
// (invité : clé = IP).
'@
AddEdit $rl $o $n


# ---------------------------------------------------------------- VERIFICATIONS (rien n'est modifie avant que tout soit OK)
foreach ($f in $Files) {
  if (-not (Test-Path $f)) { Fail "Fichier introuvable : $f" }
  if (Test-Path "$f.$Tag") { Fail "Sauvegarde deja presente ($f.$Tag). Etape deja appliquee ? Lancez d'abord -Rollback." }
}

$Cache = @{}
foreach ($f in $Files) {
  $raw = [System.IO.File]::ReadAllText($f)
  $Cache[$f] = [pscustomobject]@{
    Crlf = $raw.Contains("`r`n")
    Bom  = ([System.IO.File]::ReadAllBytes($f)[0..2] -join ',') -eq '239,187,191'
    Text = (LF $raw)
  }
}
foreach ($e in $Edits) {
  $n = ([regex]::Matches($Cache[$e.File].Text, [regex]::Escape($e.Old))).Count
  if ($n -ne $e.Count) {
    Fail "Texte a remplacer trouve $n fois (attendu $($e.Count)) dans $($e.File) :`n$($e.Old)"
  }
}
Write-Host "Verifications OK : tous les textes a remplacer existent." -ForegroundColor Green

# ---------------------------------------------------------------- SAUVEGARDE + APPLICATION
foreach ($f in $Files) { Copy-Item $f "$f.$Tag" }

foreach ($e in $Edits) {
  $Cache[$e.File].Text = $Cache[$e.File].Text.Replace($e.Old, $e.New)
}
foreach ($f in $Files) {
  $c = $Cache[$f]
  $out = $c.Text
  if ($c.Crlf) { $out = $out -replace "`n", "`r`n" }
  $enc = New-Object System.Text.UTF8Encoding($c.Bom)
  [System.IO.File]::WriteAllText($f, $out, $enc)
  Write-Host "Modifie : $f" -ForegroundColor Green
}

Write-Host ""
Write-Host "Etape 7 appliquee. Sauvegardes : *.$Tag" -ForegroundColor Cyan
Write-Host "Redemarrer / redeployer le backend, puis tester le chat en etant deconnecte ET connecte."
Write-Host "Si ca echoue encore, envoyez la ligne 'Groq error:' des logs du backend."
Write-Host "Puis dites : etape 7 validee."
